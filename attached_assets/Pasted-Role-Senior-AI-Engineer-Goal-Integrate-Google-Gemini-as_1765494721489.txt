Role: Senior AI Engineer. Goal: Integrate Google Gemini as a fallback extraction engine. Problem: Standard scraping fails on complex sites (like Oregon State). We need AI to parse the HTML when Regex returns empty fields.

Please execute the following 3 steps:

Step 1: Install Dependencies
npm install @google/generative-ai

Step 2: Create the AI Service
Create server/lib/ai-extractor.ts. This wraps the Google API.

TypeScript

// server/lib/ai-extractor.ts
import { GoogleGenerativeAI } from "@google/generative-ai";

// Ensure GEMINI_API_KEY is in your .env file
const genAI = new GoogleGenerativeAI(process.env.GEMINI_API_KEY || "");

export async function aiExtractFallback(htmlContext: string, url: string) {
  try {
    const model = genAI.getGenerativeModel({ model: "gemini-1.5-flash" });

    const prompt = `
    You are a Data Extraction Robot. Analyze this HTML snippet from a university athletics directory (${url}).
    
    Task: Extract the staff member's details.
    
    Rules:
    1. Look for Name, Title, Email, and Phone.
    2. If Title is missing, look for text nearby (siblings, parent tags) that describes a role.
    3. If Email is encrypted (Cloudflare), return null for email.
    4. Return ONLY valid JSON. No markdown.
    
    Format:
    {
      "name": "String",
      "title": "String or null",
      "email": "String or null",
      "phone": "String or null"
    }

    HTML Snippet:
    ${htmlContext.substring(0, 5000)}
    `;

    const result = await model.generateContent(prompt);
    const response = result.response.text();
    const cleanJson = response.replace(/```json|```/g, '').trim();
    
    return JSON.parse(cleanJson);
  } catch (e) {
    console.error("Gemini Extraction Failed:", e);
    return null;
  }
}
