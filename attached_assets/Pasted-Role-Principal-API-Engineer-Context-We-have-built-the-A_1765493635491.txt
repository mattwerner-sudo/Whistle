Step 2: Implement Rate Limiting & Security

Rate Limit: Apply a limit of 100 requests per 15 minutes per IP to prevent Clay/HubSpot from crashing the server.

TypeScript

// server/middleware/rate-limit.ts
import rateLimit from 'express-rate-limit';

export const apiLimiter = rateLimit({
  windowMs: 15 * 60 * 1000, // 15 minutes
  max: 100, // Limit each IP to 100 requests per windowMs
  message: { error: "Too many requests, please try again later." }
});
Step 3: Auto-Generate API Documentation (Swagger)
Create server/lib/swagger.ts and mount it at /api/docs. This allows users to import the API directly into Clay or Postman.

TypeScript

// server/lib/swagger.ts
import swaggerJsdoc from 'swagger-jsdoc';

const options = {
  definition: {
    openapi: '3.0.0',
    info: {
      title: 'Athletics Directory API',
      version: '1.0.0',
      description: 'Headless API for enriching contact data in Clay/Zapier',
    },
    servers: [{ url: '/api/v1' }],
    components: {
      securitySchemes: {
        bearerAuth: { type: 'http', scheme: 'bearer', bearerFormat: 'JWT' }
      }
    },
    security: [{ bearerAuth: [] }]
  },
  apis: ['./server/routes.ts'], // Path to the API docs
};

export const swaggerSpec = swaggerJsdoc(options);
