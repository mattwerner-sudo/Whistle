export class DatabaseStorage implements IStorage {
  async getSchoolDirectory(schoolId: string): Promise<SchoolDirectory | undefined> {
    const [directory] = await db.select().from(schoolDirectories).where(eq(schoolDirectories.schoolId, schoolId));
    return directory || undefined;
  }

  async getSchoolDirectories(options?: any): Promise<{ directories: SchoolDirectory[]; total: number }> {
     const conditions = [];
     if (options?.status) conditions.push(eq(schoolDirectories.status, options.status));
     if (options?.division) conditions.push(eq(schoolDirectories.division, options.division));
     if (options?.conference) conditions.push(eq(schoolDirectories.conference, options.conference));
     if (options?.search) {
       conditions.push(or(
         ilike(schoolDirectories.schoolName, `%${options.search}%`),
         ilike(schoolDirectories.schoolFullName, `%${options.search}%`)
       ));
     }
     const whereClause = conditions.length > 0 ? and(...conditions) : undefined;
     
     const [directories, countResult] = await Promise.all([
       db.select().from(schoolDirectories).where(whereClause).limit(options?.limit || 50).offset(options?.offset || 0),
       db.select({ count: sql<number>`count(*)` }).from(schoolDirectories).where(whereClause)
     ]);
     return { directories, total: Number(countResult[0]?.count || 0) };
  }
  
  async upsertSchoolDirectory(directory: InsertSchoolDirectory): Promise<SchoolDirectory> {
    const [result] = await db.insert(schoolDirectories).values({ ...directory, updatedAt: new Date() })
      .onConflictDoUpdate({ target: schoolDirectories.schoolId, set: { ...directory, updatedAt: new Date() } }).returning();
    return result;
  }

  async updateSchoolDirectoryStatus(schoolId: string, status: string, error?: string): Promise<void> {
    await db.update(schoolDirectories).set({ status, extractionError: error || null, updatedAt: new Date() }).where(eq(schoolDirectories.schoolId, schoolId));
  }

  async bulkUpsertSchoolDirectories(directories: InsertSchoolDirectory[]): Promise<void> {
    if (directories.length === 0) return;
    await db.insert(schoolDirectories).values(directories.map(d => ({ ...d, updatedAt: new Date() })))
      .onConflictDoNothing(); 
  }

  // UPGRADED: Staff Members Search
  async getStaffMembers(options?: {
    schoolId?: string;
    search?: string;
    division?: string;
    conference?: string;
    limit?: number;
    offset?: number;
  }): Promise<{ members: StaffMember[]; total: number }> {
    const conditions = [];

    if (options?.schoolId) conditions.push(eq(staffMembers.schoolId, options.schoolId));
    
    // UPGRADE: Full Text Search Logic
    if (options?.search) {
      const cleanSearch = options.search.replace(/[^\w\s]/g, '').trim().split(/\s+/).join(' & ');
      if (cleanSearch) {
        conditions.push(
          sql`to_tsvector('english', ${staffMembers.name} || ' ' || coalesce(${staffMembers.title}, '') || ' ' || coalesce(${staffMembers.department}, '')) @@ to_tsquery('english', ${cleanSearch + ':*'})`
        );
      }
    }

    if (options?.division) conditions.push(eq(schoolDirectories.division, options.division));
    if (options?.conference) conditions.push(eq(schoolDirectories.conference, options.conference));

    // Handle Joins if needed
    const needsJoin = options?.division || options?.conference;
    let baseQuery = db.select({
      id: staffMembers.id,
      schoolId: staffMembers.schoolId,
      name: staffMembers.name,
      title: staffMembers.title,
      email: staffMembers.email,
      phone: staffMembers.phone,
      department: staffMembers.department,
      office: staffMembers.office,
      linkedinUrl: staffMembers.linkedinUrl,
      imageUrl: staffMembers.imageUrl,
      confidence: staffMembers.confidence,
      schoolName: schoolDirectories.schoolName, 
      schoolLogo: schoolDirectories.logoUrl,    
    }).from(staffMembers);

    baseQuery = baseQuery.innerJoin(schoolDirectories, eq(staffMembers.schoolId, schoolDirectories.schoolId)) as any;

    const whereClause = conditions.length > 0 ? and(...conditions) : undefined;

    const [members, countResult] = await Promise.all([
      baseQuery
        .where(whereClause)
        .orderBy(desc(staffMembers.id))
        .limit(options?.limit || 50)
        .offset(options?.offset || 0),
      db
        .select({ count: sql<number>`count(*)` })
        .from(staffMembers)
        .innerJoin(schoolDirectories, eq(staffMembers.schoolId, schoolDirectories.schoolId))
        .where(whereClause),
    ]);

    return {
      members: members as any[], 
      total: Number(countResult[0]?.count || 0),
    };
  }

  async getStaffMember(id: number) { return (await db.select().from(staffMembers).where(eq(staffMembers.id, id)))[0]; }
  async getStaffMembersBySchool(schoolId: string) { return db.select().from(staffMembers).where(eq(staffMembers.schoolId, schoolId)); }
  
  async upsertStaffMember(member: InsertStaffMember): Promise<StaffMember> {
    const existing = await db.select().from(staffMembers).where(and(eq(staffMembers.email, member.email), eq(staffMembers.schoolId, member.schoolId)));
    if (existing.length > 0) {
      const [updated] = await db.update(staffMembers).set({ ...member, updatedAt: new Date() }).where(eq(staffMembers.id, existing[0].id)).returning();
      return updated;
    }
    const [created] = await db.insert(staffMembers).values(member).returning();
    return created;
  }
  
  async bulkUpsertStaffMembers(members: InsertStaffMember[]): Promise<void> {
    if (members.length === 0) return;
    const BATCH_SIZE = 50;
    for (let i = 0; i < members.length; i += BATCH_SIZE) {
      const batch = members.slice(i, i + BATCH_SIZE);
      await db.insert(staffMembers).values(batch)
        .onConflictDoUpdate({ 
          target: [staffMembers.email, staffMembers.schoolId], 
          set: { 
            name: sql`excluded.name`,
            title: sql`excluded.title`,
            updatedAt: new Date()
          } 
        });
    }
  }

  async deleteStaffMembersBySchool(schoolId: string): Promise<void> {
    await db.delete(staffMembers).where(eq(staffMembers.schoolId, schoolId));
  }
  
  async getStats() { 
    return { totalSchools: 0, extractedSchools: 0, totalStaff: 0, avgConfidence: 0 }; 
  }
  async logEvent(event: InsertUsageEvent) { const [e] = await db.insert(usageEvents).values(event).returning(); return e; }
  async getUsageEvents() { return { events: [], total: 0 }; }
  async getUsageStats() { return { totalEvents: 0, uniqueSessions: 0, eventsByType: {}, topSchools: [], recentActivity: [] }; }

  // NEW: Job Queue Implementation

    return db.select()
      .from(extractionJobs)
      .orderBy(desc(extractionJobs.createdAt))
      .limit(limit);
  }

// Background Worker: Processes jobs asynchronously
async function processExtractionJob(jobId: number) {
  const { storage } = await import("./storage");
  const { extractStaffFromUrl, discoverDirectoryUrl, convertToStaffMembers } = await import("./staffExtractor");
  const { getSchoolsByConference } = await import("@shared/ncaa-conferences");

  console.log(`[Job Worker] Starting job #${jobId}`);
  
  try {
    const job = await storage.getExtractionJob(jobId);
    if (!job) return;

    await storage.updateExtractionJob(jobId, { status: "processing" });

    // Determine target schools
    let targetSchools: string[] = [];
    if (job.type === 'conference' && job.targetId) {
      // In a real app, you would query the DB for schools with this conference
      // For now, we simulate finding schools associated with the conference in the DB
      const allSchools = await storage.getSchoolDirectories({ conference: job.targetId, limit: 1000 });
      targetSchools = allSchools.directories.map(s => s.schoolId);
    } else if (job.type === 'bulk') {
       const pending = await storage.getSchoolDirectories({ status: 'pending', limit: 50 });
       targetSchools = pending.directories.map(d => d.schoolId);
    } else if (job.type === 'single' && job.targetId) {
      targetSchools = [job.targetId];
    }

    if (targetSchools.length === 0) {
      await storage.updateExtractionJob(jobId, { 
        status: "completed", 
        logs: [...(job.logs || []), "No schools found matching criteria"] 
      });
      return;
    }

    await storage.updateExtractionJob(jobId, { totalSchools: targetSchools.length });

    // Process Loop
    let processedCount = 0;
    let contactsCount = 0;
    const logs: string[] = [];

    for (const schoolId of targetSchools) {
      try {
        logs.push(`Starting ${schoolId}...`);
        
        let directory = await storage.getSchoolDirectory(schoolId);
        if (!directory) {
          logs.push(`Skipped ${schoolId}: Not in DB`);
          continue;
        }

        let url = directory.directoryUrl;
        if (!url) {
          url = await discoverDirectoryUrl(directory.ncaaUrl, directory.schoolName);
          if (url) {
            await storage.upsertSchoolDirectory({ ...directory, directoryUrl: url });
          }
        }

        if (!url) {
          await storage.updateSchoolDirectoryStatus(schoolId, "no_directory");
          logs.push(`Failed ${schoolId}: No URL found`);
          continue;
        }

        const result = await extractStaffFromUrl(url);
        
        if (result.contacts.length > 0) {
          await storage.deleteStaffMembersBySchool(schoolId);
          const members = convertToStaffMembers(result.contacts, schoolId);
          await storage.bulkUpsertStaffMembers(members);
          
          await storage.upsertSchoolDirectory({
            ...directory,
            directoryUrl: url,
            status: "success",
            contactsCount: result.contacts.length,
            avgConfidence: result.diagnostics.averageConfidence,
            lastExtractedAt: new Date(),
          });
          
          contactsCount += result.contacts.length;
          logs.push(`Success ${schoolId}: ${result.contacts.length} contacts`);
        } else {
          await storage.updateSchoolDirectoryStatus(schoolId, "failed", "No contacts extracted");
          logs.push(`Warning ${schoolId}: 0 contacts found`);
        }

      } catch (err: any) {
        console.error(`Error processing ${schoolId}:`, err);
        logs.push(`Error ${schoolId}: ${err.message}`);
        await storage.updateSchoolDirectoryStatus(schoolId, "failed", err.message);
      }

      processedCount++;
      
      if (processedCount % 5 === 0 || processedCount === targetSchools.length) {
        await storage.updateExtractionJob(jobId, {
          processedSchools: processedCount,
          contactsFound: contactsCount,
          logs: logs.slice(-50) 
        });
      }

      await new Promise(r => setTimeout(r, 2000));
    }

    await storage.updateExtractionJob(jobId, {
      status: "completed",
      processedSchools: processedCount,
      contactsFound: contactsCount,
      logs: logs.slice(-50)
    });

  } catch (error: any) {
    console.error(`[Job Worker] Job #${jobId} failed:`, error);
    const { storage } = await import("./storage");
    await storage.updateExtractionJob(jobId, { 
      status: "failed", 
      logs: ["Critical Worker Error: " + error.message] 
    });
  }
}

          <div className="h-2 w-full bg-secondary rounded-full overflow-hidden">
            <div 
              className="h-full bg-primary transition-all duration-500"
              style={{ width: `${(activeJob.processedSchools / (activeJob.totalSchools || 1)) * 100}%` }}
            />
          </div>
          <div className="text-xs text-right font-medium text-primary">
            {activeJob.contactsFound} contacts found
          </div>
        </div>
      </Card>
    </div>
  );
}

export default function StaffDirectory() {
  const { toast } = useToast();
  const [selectedConf, setSelectedConf] = useState('all');
  const [searchQuery, setSearchQuery] = useState('');
  const [debouncedSearch, setDebouncedSearch] = useState('');
  
  useEffect(() => {
    const timer = setTimeout(() => setDebouncedSearch(searchQuery), 500);
    return () => clearTimeout(timer);
  }, [searchQuery]);

  const { data, isLoading } = useQuery({
    queryKey: ['/api/staff/members', selectedConf, debouncedSearch],
    queryFn: () => {
      const params = new URLSearchParams();
      if (selectedConf !== 'all') params.set('conference', selectedConf);
      if (debouncedSearch) params.set('search', debouncedSearch);
      params.set('limit', '100');
      return apiRequest('GET', `/api/staff/members?${params.toString()}`);
    }
  });

  const extractMutation = useMutation({
    mutationFn: (confId: string) => apiRequest('POST', '/api/jobs', {
      type: 'conference',
      targetId: confId
    }),
    onSuccess: () => {
      toast({ title: "Extraction Started", description: "This will run in the background." });
      queryClient.invalidateQueries({ queryKey: ['/api/jobs'] });
    }
  });

  const handleDownload = () => {
    if (!data?.members) return;
    const csv = [
      ['Name', 'Title', 'Email', 'School', 'Conference', 'Phone'].join(','),
      ...data.members.map((m: any) => [
        `"${m.name}"`, `"${m.title || ''}"`, m.email, `"${m.schoolName}"`, selectedConf, `"${m.phone || ''}"`
      ].join(','))
    ].join('\n');
    
    const blob = new Blob([csv], { type: 'text/csv' });
    const url = window.URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = `staff_export_${selectedConf}.csv`;
    a.click();
  };

  return (
    <div className="flex h-screen bg-background overflow-hidden">
      <ConferenceSidebar selected={selectedConf} onSelect={setSelectedConf} />

      <div className="flex-1 flex flex-col min-w-0">
        <header className="border-b p-4 flex items-center justify-between gap-4 bg-background/95 backdrop-blur z-10">
          <div className="flex items-center gap-4 flex-1">
            <div className="relative flex-1 max-w-md">
              <Search className="absolute left-2.5 top-2.5 h-4 w-4 text-muted-foreground" />
              <Input 
                placeholder="Search by name, title, or department..." 
                className="pl-9"
                value={searchQuery}
                onChange={(e) => setSearchQuery(e.target.value)}
              />
            </div>
          </div>
          <div className="flex items-center gap-2">
            {selectedConf !== 'all' && (
              <Button 
                variant="outline" 
                onClick={() => extractMutation.mutate(selectedConf)}
                disabled={extractMutation.isPending}
              >
                <RefreshCw className={`mr-2 h-4 w-4 ${extractMutation.isPending ? 'animate-spin' : ''}`} />
                Refresh Data
              </Button>
            )}
            <Button onClick={handleDownload} disabled={!data?.members?.length}>
              <Download className="mr-2 h-4 w-4" /> Export CSV
            </Button>
          </div>
        </header>

        <div className="flex-1 overflow-auto p-6">
          <Card className="border-0 shadow-none">
            <Table>
              <TableHeader>
                <TableRow>
                  <TableHead className="w-[300px]">Name / Title</TableHead>
                  <TableHead>School</TableHead>
                  <TableHead>Contact</TableHead>
                  <TableHead className="text-right">Action</TableHead>
                </TableRow>
              </TableHeader>
              <TableBody>
                {isLoading ? (
                  <TableRow>
                    <TableCell colSpan={4} className="h-32 text-center">
                      <Loader2 className="h-8 w-8 animate-spin mx-auto text-muted-foreground" />
                    </TableCell>
                  </TableRow>
                ) : data?.members?.length === 0 ? (
                  <TableRow>
                    <TableCell colSpan={4} className="h-32 text-center text-muted-foreground">
                      No staff found. Try selecting a conference and clicking "Refresh Data".
                    </TableCell>
                  </TableRow>
                ) : (
                  data?.members.map((member: any) => (
                    <TableRow key={member.id} className="group">
                      <TableCell>
                        <div className="flex items-center gap-3">
                          <Avatar className="h-9 w-9">
                            <AvatarImage src={member.imageUrl} />
                            <AvatarFallback>{member.name.slice(0, 2)}</AvatarFallback>
                          </Avatar>
                          <div>
                            <div className="font-medium">{member.name}</div>
                            <div className="text-xs text-muted-foreground truncate max-w-[200px]">
                              {member.title}
                            </div>
                          </div>
                        </div>
                      </TableCell>
                      <TableCell>
                        <div className="flex items-center gap-2">
                          {member.schoolLogo && <img src={member.schoolLogo} className="h-6 w-6 object-contain" />}
                          <span className="text-sm">{member.schoolName}</span>
                        </div>
                      </TableCell>
                      <TableCell>
                        <div className="flex flex-col gap-1 text-sm">
                          <a href={`mailto:${member.email}`} className="text-primary hover:underline flex items-center gap-1">
                            <Mail className="h-3 w-3" /> {member.email}
                          </a>
                          {member.phone && (
                            <div className="text-muted-foreground flex items-center gap-1 text-xs">
                              <Phone className="h-3 w-3" /> {member.phone}
                            </div>
                          )}
                        </div>
                      </TableCell>
                      <TableCell className="text-right">
                        <Button variant="ghost" size="sm" className="opacity-0 group-hover:opacity-100 transition-opacity">
                          Details <ChevronRight className="ml-1 h-3 w-3" />
                        </Button>
                      </TableCell>
                    </TableRow>
                  ))
                )}
              </TableBody>
            </Table>
          </Card>
        </div>
        
        <div className="border-t p-2 text-xs text-center text-muted-foreground bg-muted/5">
          {data?.total || 0} results found
        </div>
      </div>

      <ExtractionJobStatus />
    </div>
  );
}
