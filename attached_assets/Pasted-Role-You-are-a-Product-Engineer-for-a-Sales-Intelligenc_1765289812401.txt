  // POST Create List
  app.post("/api/lists", async (req, res) => {
    const { name } = req.body;
    const [list] = await db.insert(savedLists).values({ name, userId: 1 }).returning();
    res.json(list);
  });

  // POST Add to List
  app.post("/api/lists/:id/add", async (req, res) => {
    const { staffId } = req.body;
    await db.insert(savedListItems).values({ 
      listId: parseInt(req.params.id), 
      staffId 
    });
    res.json({ success: true });
  });
