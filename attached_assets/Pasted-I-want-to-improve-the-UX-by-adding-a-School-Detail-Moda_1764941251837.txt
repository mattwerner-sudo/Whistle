TypeScript

// client/src/pages/staff-directory.tsx

// ... (Keep your imports and helper components like ConferenceSidebar and ExtractionJobStatus) ...

export default function StaffDirectory() {
  const { toast } = useToast();
  const [selectedConf, setSelectedConf] = useState('all');
  const [searchQuery, setSearchQuery] = useState('');
  const [debouncedSearch, setDebouncedSearch] = useState('');
  
  // Modal State
  const [selectedSchoolId, setSelectedSchoolId] = useState<string | null>(null);
  
  useEffect(() => {
    const timer = setTimeout(() => setDebouncedSearch(searchQuery), 500);
    return () => clearTimeout(timer);
  }, [searchQuery]);

  // Query: Schools List
  const schoolsQuery = useQuery({
    queryKey: ['/api/staff/schools', selectedConf, debouncedSearch],
    queryFn: () => {
      const params = new URLSearchParams();
      if (selectedConf !== 'all') params.set('conference', selectedConf);
      if (debouncedSearch) params.set('search', debouncedSearch);
      params.set('limit', '100');
      return apiRequest('GET', `/api/staff/schools?${params.toString()}`);
    }
  });

  // Query: Single School Staff (Only fetches when modal is open)
  const schoolStaffQuery = useQuery({
    queryKey: ['/api/staff/members', selectedSchoolId],
    queryFn: () => apiRequest('GET', `/api/staff/members?schoolId=${selectedSchoolId}&limit=500`),
    enabled: !!selectedSchoolId
  });

  // Mutation: Extract Job
  const extractMutation = useMutation({
    mutationFn: (data: { type: string, targetId: string }) => apiRequest('POST', '/api/jobs', data),
    onSuccess: () => {
      toast({ title: "Extraction Started", description: "Running in background..." });
      queryClient.invalidateQueries({ queryKey: ['/api/jobs'] });
      // Re-fetch the school list to see status updates
      queryClient.invalidateQueries({ queryKey: ['/api/staff/schools'] });
    }
  });

  const selectedSchoolData = schoolsQuery.data?.directories?.find((s: any) => s.schoolId === selectedSchoolId);

  const getStatusColor = (status: string) => {
    switch(status) {
      case 'success': return 'bg-green-100 text-green-800 border-green-200';
      case 'processing': return 'bg-blue-100 text-blue-800 border-blue-200 animate-pulse';
      case 'failed': return 'bg-red-100 text-red-800 border-red-200';
      default: return 'bg-gray-100 text-gray-800 border-gray-200';
    }
  };

  return (
    <div className="flex h-screen bg-background overflow-hidden">
      <ConferenceSidebar selected={selectedConf} onSelect={setSelectedConf} />

      <div className="flex-1 flex flex-col min-w-0">
        {/* Header */}
        <header className="border-b p-4 flex items-center justify-between gap-4 bg-background/95 backdrop-blur z-10">
          <div className="flex items-center gap-4 flex-1">
            <div className="relative flex-1 max-w-md">
              <Search className="absolute left-2.5 top-2.5 h-4 w-4 text-muted-foreground" />
              <Input 
                placeholder="Search schools..." 
                className="pl-9"
                value={searchQuery}
                onChange={(e) => setSearchQuery(e.target.value)}
              />
            </div>
          </div>
          {selectedConf !== 'all' && (
            <Button 
              onClick={() => extractMutation.mutate({ type: 'conference', targetId: selectedConf })}
              disabled={extractMutation.isPending}
            >
              <RefreshCw className={`mr-2 h-4 w-4 ${extractMutation.isPending ? 'animate-spin' : ''}`} />
              Extract All {selectedConf}
            </Button>
          )}
        </header>

        {/* School Grid */}
        <div className="flex-1 overflow-auto bg-muted/5 p-6">
          {schoolsQuery.isLoading ? (
            <div className="flex justify-center py-20"><Loader2 className="h-8 w-8 animate-spin text-muted-foreground"/></div>
          ) : (
            <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 xl:grid-cols-4 gap-4">
              {schoolsQuery.data?.directories?.map((school: any) => (
                <Card 
                  key={school.id} 
                  className="hover:shadow-md transition-all cursor-pointer group"
                  onClick={() => setSelectedSchoolId(school.schoolId)}
                >
                  <CardHeader className="pb-3 flex flex-row items-start justify-between space-y-0">
                    <div className="flex items-center gap-3">
                      <Avatar className="h-10 w-10 border bg-white">
                        <AvatarImage src={school.logoUrl} className="object-contain p-1" />
                        <AvatarFallback>{school.schoolName.substring(0, 2)}</AvatarFallback>
                      </Avatar>
                      <div className="overflow-hidden">
                        <CardTitle className="text-base font-bold truncate" title={school.schoolName}>
                          {school.schoolName}
                        </CardTitle>
                        <div className="text-xs text-muted-foreground">{school.conference}</div>
                      </div>
                    </div>
                  </CardHeader>
                  <CardContent>
                    <div className="flex justify-between items-center text-sm">
                      <span className={`px-2 py-0.5 rounded-full text-xs border ${getStatusColor(school.status)}`}>
                        {school.status.toUpperCase()}
                      </span>
                      <span className="font-medium text-muted-foreground flex items-center gap-1">
                        <Users className="h-3 w-3" /> {school.contactsCount || 0}
                      </span>
                    </div>
                  </CardContent>
                </Card>
              ))}
            </div>
          )}
        </div>
      </div>

      <ExtractionJobStatus />

      {/* --- SCHOOL DETAIL MODAL --- */}
      <Dialog open={!!selectedSchoolId} onOpenChange={(open) => !open && setSelectedSchoolId(null)}>
        <DialogContent className="max-w-4xl max-h-[85vh] flex flex-col">
          <DialogHeader className="pb-4 border-b">
            <div className="flex items-center justify-between">
              <div className="flex items-center gap-4">
                <Avatar className="h-16 w-16 border bg-white shadow-sm">
                  <AvatarImage src={selectedSchoolData?.logoUrl} className="object-contain p-2" />
                  <AvatarFallback>SC</AvatarFallback>
                </Avatar>
                <div>
                  <DialogTitle className="text-2xl">{selectedSchoolData?.schoolName}</DialogTitle>
                  <DialogDescription className="mt-1 flex items-center gap-2">
                    <span className="font-medium text-foreground">{selectedSchoolData?.conference}</span>
                    <span>•</span>
                    <a href={selectedSchoolData?.directoryUrl} target="_blank" className="text-primary hover:underline flex items-center gap-1">
                      Visit Directory <ChevronRight className="h-3 w-3" />
                    </a>
                  </DialogDescription>
                </div>
              </div>
              <div className="flex gap-2">
                <Button 
                  variant="outline"
                  onClick={() => extractMutation.mutate({ type: 'single', targetId: selectedSchoolId! })}
                  disabled={extractMutation.isPending}
                >
                  <RefreshCw className={`mr-2 h-4 w-4 ${extractMutation.isPending ? 'animate-spin' : ''}`} />
                  Extract Data
                </Button>
              </div>
            </div>
          </DialogHeader>

          {/* Modal Content: Staff Table */}
          <div className="flex-1 overflow-auto min-h-[300px]">
            {schoolStaffQuery.isLoading ? (
              <div className="flex h-full items-center justify-center">
                <Loader2 className="h-8 w-8 animate-spin text-muted-foreground" />
              </div>
            ) : schoolStaffQuery.data?.members?.length === 0 ? (
              <div className="flex h-full flex-col items-center justify-center text-muted-foreground gap-2">
                <Users className="h-12 w-12 opacity-20" />
                <p>No staff extracted yet.</p>
                <Button variant="link" onClick={() => extractMutation.mutate({ type: 'single', targetId: selectedSchoolId! })}>
                  Run Extraction Now
                </Button>
              </div>
            ) : (
              <Table>
                <TableHeader>
                  <TableRow>
                    <TableHead>Name</TableHead>
                    <TableHead>Title</TableHead>
                    <TableHead>Email</TableHead>
                    <TableHead>Phone</TableHead>
                  </TableRow>
                </TableHeader>
                <TableBody>
                  {schoolStaffQuery.data?.members?.map((member: any) => (
                    <TableRow key={member.id}>
                      <TableCell className="font-medium">{member.name}</TableCell>
                      <TableCell className="text-muted-foreground text-sm">{member.title}</TableCell>
                      <TableCell>
                        <a href={`mailto:${member.email}`} className="text-primary hover:underline text-sm flex items-center gap-1">
                          <Mail className="h-3 w-3" /> {member.email}
                        </a>
                      </TableCell>
                      <TableCell className="text-sm text-muted-foreground">
                        {member.phone || '-'}
                      </TableCell>
                    </TableRow>
                  ))}
                </TableBody>
              </Table>
            )}
          </div>
          
          <DialogFooter className="border-t pt-4 text-xs text-muted-foreground">
            {schoolStaffQuery.data?.members?.length || 0} contacts found • {selectedSchoolData?.ncaaUrl}
          </DialogFooter>
        </DialogContent>
      </Dialog>
    </div>
  );
}
