Step 4: Build the "Webhook Dispatcher"
Create server/lib/webhooks.ts. This function will be called whenever your scraper detects a change (New Hire), pushing that data out to Zapier.

TypeScript

// server/lib/webhooks.ts
import { db } from "../db";
import { webhookSubscriptions } from "@shared/schema";
import crypto from "crypto";

export async function dispatchWebhook(userId: number, eventType: string, payload: any) {
  const subs = await db.select().from(webhookSubscriptions).where(eq(webhookSubscriptions.userId, userId));

  for (const sub of subs) {
    if (sub.eventTypes?.includes(eventType)) {
      try {
        // Sign payload for security
        const signature = crypto.createHmac('sha256', sub.secret).update(JSON.stringify(payload)).digest('hex');
        
        await fetch(sub.url, {
          method: 'POST',
          headers: {
            'Content-Type': 'application/json',
            'X-Signature': signature
          },
          body: JSON.stringify({ event: eventType, data: payload })
        });
      } catch (err) {
        console.error(`Webhook failed for ${sub.url}`, err);
      }
    }
  }
}
