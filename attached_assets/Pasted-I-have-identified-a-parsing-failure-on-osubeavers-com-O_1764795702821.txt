Sidearm sites often include newlines (\n) and tabs (\t) inside titles. Add this helper function at the top of the file and use it on the title result.

function cleanSidearmText(text: string): string {
  return text
    .replace(/[\n\r\t]+/g, ' ') // Replace newlines with spaces
    .replace(/\s\s+/g, ' ')     // Collapse multiple spaces
    .trim();
}
