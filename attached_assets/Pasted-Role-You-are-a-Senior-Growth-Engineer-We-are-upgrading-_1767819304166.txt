// server/lib/graph-engine.ts
import { db } from "../db";
import { careerHistory, staffMembers } from "@shared/schema";
import { eq, and, inArray } from "drizzle-orm";

export async function findWarmPaths(targetSchoolId: string, customerSchoolIds: string[]) {
  // 1. Get all staff at target school
  const targetStaff = await db.select().from(staffMembers).where(eq(staffMembers.schoolId, targetSchoolId));
  const targetStaffIds = targetStaff.map(s => s.id);

  // 2. Check their history for matches with customer schools
  const warmLeads = await db.select()
    .from(careerHistory)
    .where(and(
      inArray(careerHistory.staffId, targetStaffIds),
      inArray(careerHistory.schoolId, customerSchoolIds)
    ));

  return warmLeads.map(lead => ({
    staffId: lead.staffId,
    previousSchool: lead.schoolId,
    signal: "Former User"
  }));
}
