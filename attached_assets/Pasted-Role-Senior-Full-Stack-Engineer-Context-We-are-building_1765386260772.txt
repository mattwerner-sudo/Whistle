Step 1: Create the "Bulk Fetch" Endpoint
Update server/routes.ts. We need an endpoint that accepts a large list of School IDs and returns all their staff. Standard GET requests might hit URL length limits, so we use POST.

TypeScript

// server/routes.ts

  // POST /api/staff/bulk-fetch
  // Payload: { schoolIds: ["alabama", "ohiostate", ...] }
  app.post("/api/staff/bulk-fetch", async (req, res) => {
    try {
      const { schoolIds } = req.body;
      
      if (!schoolIds || !Array.isArray(schoolIds) || schoolIds.length === 0) {
        return res.json({ members: [], count: 0 });
      }

      // Use the storage layer to fetch via "IN" clause
      // Note: You may need to add getStaffBySchoolIds to storage.ts
      const { db } = await import("./db");
      const { staffMembers, schoolDirectories } = await import("@shared/schema");
      const { inArray, eq } = await import("drizzle-orm");
      
      const members = await db
        .select({
          // Select standard fields + school info
          id: staffMembers.id,
          name: staffMembers.name,
          title: staffMembers.title,
          email: staffMembers.email,
          phone: staffMembers.phone,
          schoolName: schoolDirectories.schoolName,
          schoolId: schoolDirectories.schoolId
        })
        .from(staffMembers)
        .innerJoin(schoolDirectories, eq(staffMembers.schoolId, schoolDirectories.schoolId))
        .where(inArray(staffMembers.schoolId, schoolIds))
        .limit(5000); // Safety limit

      res.json({ members, count: members.length });
    } catch (error: any) {
      res.status(500).json({ error: error.message });
    }
  });
