// server/staffExtractor.ts

function detectTechStack($: cheerio.CheerioAPI): string[] {
  const tech = new Set<string>();
  const html = $.html();

  // 1. Detect CMS / Website Provider
  if (html.includes('sidearmsports') || $('a[href*="sidearmsports.com"]').length) tech.add('Sidearm Sports');
  if (html.includes('neulion')) tech.add('Neulion');
  if (html.includes('wmt.digital')) tech.add('WMT Digital');

  // 2. Detect Ticketing (often in navigation links)
  if ($('a[href*="ticketmaster"]').length) tech.add('Ticketmaster');
  if ($('a[href*="paciolan"]').length) tech.add('Paciolan');
  if ($('a[href*="fevo"]').length) tech.add('Fevo');

  // 3. Detect Logistics/Ops (often in footers or vendor links)
  if (html.includes('teamworks')) tech.add('Teamworks');
  if (html.includes('arms software')) tech.add('ARMS');
  if (html.includes('catapult')) tech.add('Catapult');

  return Array.from(tech);
}
