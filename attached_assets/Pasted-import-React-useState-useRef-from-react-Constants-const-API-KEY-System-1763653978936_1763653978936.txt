import React, { useState, useRef } from 'react';

// --- Constants ---
const API_KEY = ""; // System provided key will work here at runtime

// --- Helper: Simple Markdown Parser (to avoid external dependencies for a beginner) ---
const SimpleMarkdown = ({ text }) => {
  if (!text) return null;
  // Convert **text** to bold
  const html = text
    .replace(/\*\*(.*?)\*\*/g, '<strong>$1</strong>')
    .replace(/\n/g, '<br />');
  return <div dangerouslySetInnerHTML={{ __html: html }} />;
};

export default function App() {
  // --- State ---
  const [activeTab, setActiveTab] = useState('url');
  const [urlInput, setUrlInput] = useState('https://floridagators.com/staff-directory');
  const [textInput, setTextInput] = useState('');
  const [extractedData, setExtractedData] = useState([]);
  const [loading, setLoading] = useState(false);
  const [loadingMessage, setLoadingMessage] = useState('Processing...');
  
  // Modal States
  const [emailModalOpen, setEmailModalOpen] = useState(false);
  const [prepModalOpen, setPrepModalOpen] = useState(false);
  const [analysisOpen, setAnalysisOpen] = useState(false);
  const [currentRecipient, setCurrentRecipient] = useState(null);
  
  // AI Content States
  const [aiResponse, setAiResponse] = useState('');
  const [emailDraft, setEmailDraft] = useState({ subject: '', body: '' });
  const [emailContext, setEmailContext] = useState('');
  const [prepTopic, setPrepTopic] = useState('');
  const [isAiGenerating, setIsAiGenerating] = useState(false);

  // --- Logic: Fetching ---
  const fetchFromUrl = async () => {
    if (!urlInput) return alert("Please enter a URL");
    setLoading(true);
    setLoadingMessage('Fetching data...');
    setExtractedData([]);

    const strategies = [
      {
        name: 'AllOrigins',
        url: `https://api.allorigins.win/get?url=${encodeURIComponent(urlInput)}`,
        handler: async (res) => {
          const data = await res.json();
          return data.contents;
        }
      },
      {
        name: 'CodeTabs',
        url: `https://api.codetabs.com/v1/proxy?quest=${encodeURIComponent(urlInput)}`,
        handler: async (res) => res.text()
      }
    ];

    let content = null;

    for (const strategy of strategies) {
      try {
        console.log(`Attempting fetch via ${strategy.name}...`);
        const response = await fetch(strategy.url);
        if (response.ok) {
          content = await strategy.handler(response);
          if (content && content.length > 100) break; 
        }
      } catch (err) {
        console.warn(`${strategy.name} failed:`, err);
      }
    }

    if (content) {
      parseHtml(content);
    } else {
      alert("Could not auto-fetch. Please switch to 'Paste HTML' tab.");
      setLoading(false);
    }
  };

  // --- Logic: Parsing ---
  const parseHtml = (htmlString) => {
    setLoadingMessage('Parsing HTML...');
    const parser = new DOMParser();
    const doc = parser.parseFromString(htmlString, 'text/html');
    const newData = [];
    const mailtoLinks = doc.querySelectorAll('a[href^="mailto:"]');
    const processedEmails = new Set();

    mailtoLinks.forEach(link => {
      const email = link.href.replace('mailto:', '').trim();
      if (!email || processedEmails.has(email)) return;
      processedEmails.add(email);

      let container = link.closest('tr') || link.closest('li') || link.closest('.staff-member') || link.closest('.card') || link.parentElement.parentElement;
      if (container) {
        const person = extractPersonFromContainer(container, email);
        newData.push(person);
      }
    });

    setExtractedData(newData);
    setLoading(false);
  };

  const extractPersonFromContainer = (container, email) => {
    const textContent = container.innerText.replace(/\s+/g, ' ').trim();
    const phoneMatch = textContent.match(/(\(?\d{3}\)?[\s.-]?\d{3}[\s.-]?\d{4})/);
    let phone = phoneMatch ? phoneMatch[0] : "";
    let name = "Unknown";

    // Name extraction logic
    const bioLink = container.querySelector('a[aria-label*="Bio"], a[title*="Bio"]');
    if (bioLink) {
        const bioText = bioLink.getAttribute('aria-label') || bioLink.getAttribute('title') || bioLink.innerText;
        const bioMatch = bioText.match(/Full Bio for (.+)/i);
        if (bioMatch) name = bioMatch[1];
    }
    
    if (name === "Unknown") {
        const allLinks = container.querySelectorAll('a');
        for (let a of allLinks) {
                const txt = a.innerText || a.getAttribute('aria-label') || "";
                const match = txt.match(/Full Bio for (.+)/i);
                if (match) {
                    name = match[1];
                    break;
                }
        }
    }

    if (name === "Unknown") {
        const nameCandidates = container.querySelectorAll('h1, h2, h3, h4, h5, strong, b, .name');
        if (nameCandidates.length > 0) name = nameCandidates[0].innerText.trim();
    }

    if (name === "Unknown" && container.tagName === 'TR') {
            const cells = container.querySelectorAll('td');
            if (cells.length > 0) name = cells[0].innerText.trim();
    }
    name = name.replace(/Full Bio for/i, '').trim();

    // Title extraction logic
    let title = "";
    const titleCandidates = container.querySelectorAll('.title, .position');
    if (titleCandidates.length > 0) {
        title = titleCandidates[0].innerText.trim();
    } 
    
    if ((!title || title.length < 2) && name !== "Unknown") {
        let cleanText = textContent.replace(name, "").replace(email, "").replace(phone, "").replace(/Full Bio for/gi, "").replace(/Phone/gi, "").replace(/Email/gi, "").trim();
        if (cleanText.length > 2) {
                const potentialTitle = cleanText.split(/[\n,]/)[0].trim();
                if (potentialTitle.length > 2 && potentialTitle.length < 60) title = potentialTitle;
        }
    }
    
    if ((!title || title === "Unknown") && container.tagName === 'TR') {
        const cells = container.querySelectorAll('td');
        if (cells.length >= 2) title = cells[1].innerText.trim();
    }

    return { name, title, email, phone };
  };

  // --- Logic: AI Calls ---
  const callGemini = async (prompt) => {
    const url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-09-2025:generateContent?key=${API_KEY}`;
    const payload = { contents: [{ parts: [{ text: prompt }] }] };
    
    try {
      const response = await fetch(url, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(payload)
      });
      const data = await response.json();
      return data.candidates?.[0]?.content?.parts?.[0]?.text || "Error: No response.";
    } catch (e) {
      console.error(e);
      return "Error calling API.";
    }
  };

  const handleAnalyze = async () => {
    if (extractedData.length === 0) return;
    setAnalysisOpen(true);
    setAiResponse('');
    setIsAiGenerating(true);

    const dataSummary = extractedData.slice(0, 100).map(p => `${p.name} (${p.title})`).join("\n");
    const prompt = `Analyze this staff list. Provide a summary with: 1. Hierarchy Overview, 2. Key Roles, 3. Department Breakdown, 4. Insights. List:\n${dataSummary}`;
    
    const result = await callGemini(prompt);
    setAiResponse(result);
    setIsAiGenerating(false);
  };

  const handleCleanData = async () => {
    setIsAiGenerating(true);
    const subset = extractedData.slice(0, 20);
    const prompt = `Fix this JSON contact data: Remove "Full Bio for", capitalize properly, format phones. Return ONLY valid JSON array. Input:\n${JSON.stringify(subset)}`;
    
    try {
        let result = await callGemini(prompt);
        result = result.replace(/```json/g, '').replace(/```/g, '').trim();
        const cleaned = JSON.parse(result);
        
        const updatedData = [...extractedData];
        cleaned.forEach((c, i) => {
            if (updatedData[i]) updatedData[i] = { ...updatedData[i], ...c };
        });
        setExtractedData(updatedData);
        alert(`Cleaned top ${cleaned.length} rows.`);
    } catch (e) {
        alert("Cleanup failed.");
    }
    setIsAiGenerating(false);
  };

  const handleGenerateEmail = async () => {
    setIsAiGenerating(true);
    const prompt = `Write a cold email. Recipient: ${currentRecipient.name}, Title: ${currentRecipient.title}, Goal: ${emailContext || "Intro"}. Output JSON: { "subject": "...", "body": "..." }. No markdown.`;
    
    try {
        let result = await callGemini(prompt);
        result = result.replace(/```json/g, '').replace(/```/g, '').trim();
        const json = JSON.parse(result);
        setEmailDraft(json);
    } catch (e) {
        alert("Generation failed");
    }
    setIsAiGenerating(false);
  };

  const handleGeneratePrep = async () => {
    if (!prepTopic) return alert("Enter a topic");
    setIsAiGenerating(true);
    setAiResponse(''); // Reuse aiResponse for prep content
    const prompt = `Meeting with ${currentRecipient.name} (${currentRecipient.title}). Topic: ${prepTopic}. Provide markdown summary: Role Context, Agenda (3 bullets), Strategic Questions (2), Icebreaker.`;
    
    const result = await callGemini(prompt);
    setAiResponse(result);
    setIsAiGenerating(false);
  };

  // --- Logic: CSV Export ---
  const downloadCSV = () => {
    if (extractedData.length === 0) return;
    const headers = ["Name", "Title", "Email", "Phone"];
    const csvRows = [headers.join(",")];
    extractedData.forEach(row => {
        const values = [
            `"${(row.name || '').replace(/"/g, '""')}"`,
            `"${(row.title || '').replace(/"/g, '""')}"`,
            `"${row.email}"`,
            `"${row.phone}"`
        ];
        csvRows.push(values.join(","));
    });
    const link = document.createElement("a");
    link.href = "data:text/csv;charset=utf-8," + encodeURI(csvRows.join("\n"));
    link.download = "staff_data.csv";
    document.body.appendChild(link);
    link.click();
    document.body.removeChild(link);
  };

  // --- Render ---
  return (
    <div className="bg-gray-50 min-h-screen font-sans text-gray-800">
      {/* Header */}
      <header className="bg-indigo-700 text-white shadow-lg">
        <div className="container mx-auto px-6 py-4 flex justify-between items-center">
            <div className="flex items-center space-x-3">
                <div className="bg-white/10 p-2 rounded-lg">
                    {/* Icon placeholder - FontAwesome usually handled in index.html */}
                    <span className="text-xl font-bold">📂</span>
                </div>
                <h1 className="text-xl font-bold tracking-tight">Directory Extractor <span className="text-indigo-300 font-light">+ Gemini AI</span></h1>
            </div>
            <div className="text-xs bg-indigo-800 px-3 py-1 rounded-full border border-indigo-600">
                ✨ AI Enabled
            </div>
        </div>
      </header>

      <main className="container mx-auto px-4 py-8">
        <div className="grid grid-cols-1 lg:grid-cols-12 gap-8">
            
            {/* Left Panel */}
            <div className="lg:col-span-4 space-y-6">
                <div className="bg-white rounded-xl shadow-sm border border-gray-200 p-6">
                    <h2 className="text-lg font-semibold mb-4">1. Get Data</h2>
                    <div className="flex border-b border-gray-200 mb-4">
                        <button onClick={() => setActiveTab('url')} className={`flex-1 py-2 text-sm font-medium transition-colors ${activeTab === 'url' ? 'text-indigo-600 border-b-2 border-indigo-600' : 'text-gray-500'}`}>From URL</button>
                        <button onClick={() => setActiveTab('text')} className={`flex-1 py-2 text-sm font-medium transition-colors ${activeTab === 'text' ? 'text-indigo-600 border-b-2 border-indigo-600' : 'text-gray-500'}`}>Paste HTML</button>
                    </div>

                    {activeTab === 'url' ? (
                        <div className="space-y-3">
                            <label className="block text-sm font-medium text-gray-700">Directory URL</label>
                            <input type="text" value={urlInput} onChange={(e) => setUrlInput(e.target.value)} className="w-full px-4 py-2 border border-gray-300 rounded-lg focus:ring-2 focus:ring-indigo-500 outline-none" />
                            <button onClick={fetchFromUrl} disabled={loading} className="w-full bg-indigo-600 hover:bg-indigo-700 text-white font-medium py-2 px-4 rounded-lg transition-all">
                                {loading ? 'Loading...' : 'Fetch & Extract'}
                            </button>
                        </div>
                    ) : (
                        <div className="space-y-3">
                            <textarea value={textInput} onChange={(e) => setTextInput(e.target.value)} rows="8" className="w-full p-3 border border-gray-300 rounded-lg focus:ring-2 focus:ring-indigo-500 text-xs font-mono" placeholder="Paste HTML source here..." />
                            <button onClick={() => parseHtml(textInput)} className="w-full bg-gray-800 hover:bg-gray-900 text-white font-medium py-2 px-4 rounded-lg">Process HTML</button>
                        </div>
                    )}
                </div>

                {/* AI Tools */}
                {extractedData.length > 0 && (
                    <div className="bg-gradient-to-br from-indigo-50 to-white rounded-xl shadow-sm border border-indigo-100 p-6">
                        <h2 className="text-lg font-semibold mb-2">Gemini AI Tools</h2>
                        <div className="space-y-3">
                            <button onClick={handleAnalyze} disabled={isAiGenerating} className="w-full bg-white hover:bg-indigo-50 text-indigo-700 font-medium py-3 px-4 rounded-lg border border-indigo-200 flex items-center justify-between">
                                <span>📊 Analyze Team Structure</span>
                            </button>
                            <button onClick={handleCleanData} disabled={isAiGenerating} className="w-full bg-white hover:bg-indigo-50 text-indigo-700 font-medium py-3 px-4 rounded-lg border border-indigo-200 flex items-center justify-between">
                                <span>🧹 Smart Clean & Fix</span>
                            </button>
                        </div>
                    </div>
                )}

                {/* Stats */}
                {extractedData.length > 0 && (
                    <div className="bg-white rounded-xl shadow-sm border border-gray-200 p-6">
                         <div className="grid grid-cols-2 gap-4 text-center">
                            <div className="p-3 bg-gray-50 rounded-lg">
                                <div className="text-2xl font-bold">{extractedData.length}</div>
                                <div className="text-xs text-gray-500">Contacts</div>
                            </div>
                            <div className="p-3 bg-green-50 rounded-lg">
                                <div className="text-2xl font-bold text-green-700">{extractedData.filter(d => d.email).length}</div>
                                <div className="text-xs text-green-600">Emails</div>
                            </div>
                        </div>
                        <button onClick={downloadCSV} className="mt-4 w-full bg-green-600 hover:bg-green-700 text-white font-medium py-2 rounded-lg">Download CSV</button>
                    </div>
                )}
            </div>

            {/* Right Panel: Results */}
            <div className="lg:col-span-8">
                <div className="bg-white rounded-xl shadow-sm border border-gray-200 min-h-[600px] flex flex-col">
                    {loading ? (
                        <div className="flex-grow flex flex-col items-center justify-center text-gray-500">
                            <div className="animate-spin rounded-full h-12 w-12 border-b-2 border-indigo-500 mb-4"></div>
                            <p>{loadingMessage}</p>
                        </div>
                    ) : extractedData.length === 0 ? (
                        <div className="flex-grow flex flex-col items-center justify-center text-gray-400 p-12 text-center">
                            <p className="text-6xl mb-4">📋</p>
                            <h3 className="text-xl font-semibold text-gray-600">Ready to Extract</h3>
                            <p className="text-sm mt-2">Enter a URL to begin.</p>
                        </div>
                    ) : (
                        <div className="overflow-x-auto">
                            <table className="w-full text-left border-collapse">
                                <thead>
                                    <tr className="bg-gray-50 text-gray-600 text-xs uppercase tracking-wider border-b border-gray-200">
                                        <th className="p-4 font-semibold">Name</th>
                                        <th className="p-4 font-semibold">Title</th>
                                        <th className="p-4 font-semibold">Email</th>
                                        <th className="p-4 font-semibold text-right">Actions</th>
                                    </tr>
                                </thead>
                                <tbody className="divide-y divide-gray-100 text-sm">
                                    {extractedData.map((person, idx) => (
                                        <tr key={idx} className="hover:bg-gray-50 group">
                                            <td className="p-4 font-medium">{person.name}</td>
                                            <td className="p-4 text-gray-600 max-w-xs truncate" title={person.title}>{person.title}</td>
                                            <td className="p-4 text-indigo-600 truncate">{person.email}</td>
                                            <td className="p-4 text-right">
                                                <div className="flex justify-end space-x-2">
                                                    <button 
                                                        onClick={() => { setCurrentRecipient(person); setEmailModalOpen(true); setEmailDraft({subject:'', body:''}); }}
                                                        className="text-xs bg-white border border-indigo-200 text-indigo-600 px-2 py-1 rounded hover:bg-indigo-50">
                                                        ✉️ Draft
                                                    </button>
                                                    <button 
                                                        onClick={() => { setCurrentRecipient(person); setPrepModalOpen(true); setAiResponse(''); }}
                                                        className="text-xs bg-white border border-emerald-200 text-emerald-600 px-2 py-1 rounded hover:bg-emerald-50">
                                                        📝 Prep
                                                    </button>
                                                </div>
                                            </td>
                                        </tr>
                                    ))}
                                </tbody>
                            </table>
                        </div>
                    )}
                </div>
            </div>
        </div>

        {/* Analysis Modal */}
        {analysisOpen && (
            <div className="fixed inset-0 bg-black/50 backdrop-blur-sm z-50 flex items-center justify-center p-4">
                <div className="bg-white rounded-xl w-full max-w-3xl max-h-[80vh] flex flex-col">
                    <div className="p-6 border-b flex justify-between items-center">
                        <h3 className="text-xl font-bold text-indigo-900">Department Analysis</h3>
                        <button onClick={() => setAnalysisOpen(false)} className="text-gray-500 hover:text-black">✕</button>
                    </div>
                    <div className="p-6 overflow-y-auto">
                        {isAiGenerating ? <p className="text-center animate-pulse">Analyzing...</p> : <SimpleMarkdown text={aiResponse} />}
                    </div>
                </div>
            </div>
        )}

        {/* Email Modal */}
        {emailModalOpen && currentRecipient && (
             <div className="fixed inset-0 bg-black/50 backdrop-blur-sm z-50 flex items-center justify-center p-4">
                <div className="bg-white rounded-xl w-full max-w-2xl max-h-[90vh] flex flex-col">
                    <div className="bg-indigo-600 p-6 text-white rounded-t-xl flex justify-between">
                        <h3 className="text-xl font-bold">✨ Draft Email for {currentRecipient.name}</h3>
                        <button onClick={() => setEmailModalOpen(false)} className="text-white/70 hover:text-white">✕</button>
                    </div>
                    <div className="p-6 overflow-y-auto space-y-4">
                        <div>
                            <label className="block text-sm font-medium text-gray-700 mb-1">Goal & Context</label>
                            <textarea 
                                value={emailContext} 
                                onChange={(e) => setEmailContext(e.target.value)}
                                className="w-full border rounded p-2 text-sm" 
                                placeholder="e.g. I want to discuss a partnership..."
                            />
                        </div>
                        <button onClick={handleGenerateEmail} disabled={isAiGenerating} className="w-full bg-indigo-600 text-white py-2 rounded hover:bg-indigo-700">
                            {isAiGenerating ? 'Generating...' : 'Generate Draft'}
                        </button>
                        {emailDraft.subject && (
                            <div className="bg-gray-50 p-4 rounded border mt-4">
                                <div className="mb-2 font-bold">Subject: {emailDraft.subject}</div>
                                <pre className="whitespace-pre-wrap font-sans text-sm">{emailDraft.body}</pre>
                            </div>
                        )}
                    </div>
                </div>
            </div>
        )}

        {/* Prep Modal */}
        {prepModalOpen && currentRecipient && (
             <div className="fixed inset-0 bg-black/50 backdrop-blur-sm z-50 flex items-center justify-center p-4">
                <div className="bg-white rounded-xl w-full max-w-2xl max-h-[90vh] flex flex-col">
                    <div className="bg-emerald-600 p-6 text-white rounded-t-xl flex justify-between">
                        <h3 className="text-xl font-bold">✨ Meeting Prep: {currentRecipient.name}</h3>
                        <button onClick={() => setPrepModalOpen(false)} className="text-white/70 hover:text-white">✕</button>
                    </div>
                    <div className="p-6 overflow-y-auto space-y-4">
                        <div>
                            <label className="block text-sm font-medium text-gray-700 mb-1">Meeting Topic</label>
                            <input 
                                type="text"
                                value={prepTopic} 
                                onChange={(e) => setPrepTopic(e.target.value)}
                                className="w-full border rounded p-2 text-sm" 
                                placeholder="e.g. Sales demo..."
                            />
                        </div>
                        <button onClick={handleGeneratePrep} disabled={isAiGenerating} className="w-full bg-emerald-600 text-white py-2 rounded hover:bg-emerald-700">
                            {isAiGenerating ? 'Thinking...' : 'Generate Intelligence'}
                        </button>
                        {aiResponse && !isAiGenerating && (
                            <div className="bg-emerald-50 p-4 rounded border mt-4">
                                <SimpleMarkdown text={aiResponse} />
                            </div>
                        )}
                    </div>
                </div>
            </div>
        )}

      </main>
    </div>
  );
}
