  // POST /api/growth/match-accounts
  app.post("/api/growth/match-accounts", async (req, res) => {
    try {
      const { accountNames } = req.body; // Array of strings e.g. ["Alabama", "Ohio St"]
      if (!accountNames || !Array.isArray(accountNames)) return res.status(400).send("Invalid input");

      const { db } = await import("./db");
      const { staffMembers } = await import("@shared/schema");
      const { inArray } = await import("drizzle-orm");

      const results = [];
      const matchedIds = new Set<string>();

      // 1. Fuzzy Match Phase
      for (const name of accountNames) {
        const match = fuzzysort.go(name, searchIndex, { key: 'searchStr', limit: 1 })[0];
        if (match) {
          results.push({ input: name, match: match.obj.schoolName, id: match.obj.id });
          matchedIds.add(match.obj.id);
        } else {
          results.push({ input: name, match: null, id: null });
        }
      }

      // 2. Enrichment Phase (Get contacts for matches)
      let contacts: any[] = [];
      if (matchedIds.size > 0) {
        contacts = await db.select().from(staffMembers).where(inArray(staffMembers.schoolId, Array.from(matchedIds)));
      }

      res.json({
        summary: {
          inputCount: accountNames.length,
          matchedCount: matchedIds.size,
          contactsFound: contacts.length
        },
        matches: results,
        contacts: contacts // The "Gold" for the marketer
      });

    } catch (e: any) {
      res.status(500).json({ error: e.message });
    }
  });
Step 3: Create the "New Hires" API
Update server/routes.ts. Add an endpoint to fetch staff detected recently.

TypeScript

  // GET /api/growth/new-hires
  app.get("/api/growth/new-hires", async (req, res) => {
    const { db } = await import("./db");
    const { staffMembers, schoolDirectories } = await import("@shared/schema");
    const { desc, gt, eq } = await import("drizzle-orm");

    // Fetch staff 'firstSeenAt' in last 7 days
    const sevenDaysAgo = new Date();
    sevenDaysAgo.setDate(sevenDaysAgo.getDate() - 7);

    const hires = await db.select({
      name: staffMembers.name,
      title: staffMembers.title,
      school: schoolDirectories.schoolName,
      email: staffMembers.email,
      detectedAt: staffMembers.firstSeenAt
    })
    .from(staffMembers)
    .innerJoin(schoolDirectories, eq(staffMembers.schoolId, schoolDirectories.schoolId))
    .where(gt(staffMembers.firstSeenAt, sevenDaysAgo))
    .orderBy(desc(staffMembers.firstSeenAt))
    .limit(50);

    res.json(hires);
  });

// client/src/pages/growth-tools.tsx
import { useState } from 'react';
import { useQuery, useMutation } from '@tanstack/react-query';
import { apiRequest } from '@/lib/queryClient';
import { Button } from '@/components/ui/button';
import { Textarea } from '@/components/ui/textarea';
import { Card, CardHeader, CardTitle, CardContent } from '@/components/ui/card';
import { Badge } from '@/components/ui/badge';
import { Download, Rocket, Users, Building } from 'lucide-react';

export default function GrowthTools() {
  const [input, setInput] = useState('');
  const [matchResult, setMatchResult] = useState<any>(null);

  // New Hires Query
  const hiresQuery = useQuery({
    queryKey: ['/api/growth/new-hires'],
    queryFn: () => apiRequest('GET', '/api/growth/new-hires')
  });

  // Match Mutation
  const matchMutation = useMutation({
    mutationFn: (names: string[]) => apiRequest('POST', '/api/growth/match-accounts', { accountNames: names }),
    onSuccess: (data) => setMatchResult(data)
  });

  const handleEnrich = () => {
    const names = input.split('\n').filter(n => n.trim().length > 0);
    matchMutation.mutate(names);
  };

  const downloadCSV = () => {
    if (!matchResult?.contacts) return;
    const csv = ["Name,Title,Email,School,Phone"].join(',') + '\n' +
      matchResult.contacts.map((c: any) => `${c.name},"${c.title}",${c.email},${c.schoolId},${c.phone}`).join('\n');
    const blob = new Blob([csv], { type: 'text/csv' });
    const url = window.URL.createObjectURL(blob);
    const a = document.createElement('a'); a.href = url; a.download = "enriched_contacts.csv"; a.click();
  };

  return (
    <div className="p-6 grid grid-cols-1 md:grid-cols-2 gap-6 h-screen bg-muted/5 overflow-auto">
      
      {/* ABM LIST MATCHER */}
      <Card className="h-fit">
        <CardHeader>
          <CardTitle className="flex items-center gap-2">
            <Building className="h-5 w-5 text-blue-600" /> 
            Account Enrichment
          </CardTitle>
        </CardHeader>
        <CardContent className="space-y-4">
          <p className="text-sm text-muted-foreground">Paste target schools (one per line) to extract staff.</p>
          <Textarea 
            rows={8} 
            placeholder="University of Alabama&#10;Ohio State&#10;Texas"
            value={input}
            onChange={e => setInput(e.target.value)}
          />
          <Button className="w-full" onClick={handleEnrich} disabled={matchMutation.isPending}>
            {matchMutation.isPending ? 'Matching...' : 'Match & Enrich'}
          </Button>

          {matchResult && (
            <div className="rounded-lg bg-green-50 p-4 border border-green-100">
              <div className="flex justify-between items-center mb-2">
                <span className="font-bold text-green-800">Success!</span>
                <Badge variant="outline" className="bg-white">{matchResult.contacts.length} Contacts Found</Badge>
              </div>
              <div className="text-xs text-green-700 mb-4">
                Matched {matchResult.summary.matchedCount} of {matchResult.summary.inputCount} schools.
              </div>
              <Button variant="outline" size="sm" className="w-full bg-white" onClick={downloadCSV}>
                <Download className="mr-2 h-4 w-4" /> Download CSV
              </Button>
            </div>
          )}
        </CardContent>
      </Card>

      {/* NEW HIRES FEED */}
      <Card className="h-fit">
        <CardHeader>
          <CardTitle className="flex items-center gap-2">
            <Rocket className="h-5 w-5 text-orange-600" /> 
            New Hires (Last 7 Days)
          </CardTitle>
        </CardHeader>
        <CardContent>
          {hiresQuery.isLoading ? <p>Loading...</p> : (
            <div className="space-y-4">
              {hiresQuery.data?.map((hire: any, i: number) => (
                <div key={i} className="flex items-start gap-3 pb-3 border-b last:border-0">
                  <div className="h-8 w-8 rounded-full bg-orange-100 flex items-center justify-center text-orange-700 text-xs font-bold">
                    {hire.name.charAt(0)}
                  </div>
                  <div>
                    <div className="font-medium text-sm">{hire.name}</div>
                    <div className="text-xs text-muted-foreground">{hire.title}</div>
                    <div className="text-xs font-semibold text-blue-600 mt-0.5">{hire.school}</div>
                  </div>
                </div>
              ))}
              {hiresQuery.data?.length === 0 && <p className="text-muted-foreground text-sm">No new hires detected this week.</p>}
            </div>
          )}
        </CardContent>
      </Card>
    </div>
  );
}
