Please execute the following 3 steps to implement Slack Alerts and the "Click-to-Reveal" UX pattern:

Step 1: Create the Notification Service
Create a new file server/lib/notifications.ts. This is a lightweight wrapper to send alerts to Slack.

TypeScript

// server/lib/notifications.ts

// You will need to add your Webhook URL in Replit Secrets as SLACK_WEBHOOK_URL
const SLACK_WEBHOOK = process.env.SLACK_WEBHOOK_URL;

export async function sendSlackAlert(message: string, type: 'info' | 'success' | 'alert' = 'info') {
  if (!SLACK_WEBHOOK) {
    console.log(`[Slack Mock] ${message}`);
    return;
  }

  const icons = { info: 'ℹ️', success: '✅', alert: '🚨' };
  
  try {
    await fetch(SLACK_WEBHOOK, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        text: `${icons[type]} *Athletics Directory Alert*\n${message}`
      })
    });
  } catch (err) {
    console.error("Failed to send Slack alert:", err);
  }
}

  // NEW: Fire Slack Alert for high-value actions
  if (eventType === 'export') {
    const count = details?.count || 0;
    const format = details?.format || 'csv';
    await sendSlackAlert(`User (IP: ${hashIp(req.ip)}) just exported **${count} contacts** in ${format} format.`, 'success');
  }
  
  if (eventType === 'reveal_contact') {
    const contactName = details?.contactName || 'Unknown';
    await sendSlackAlert(`User revealed contact info for: **${contactName}** at ${schoolName || 'Unknown School'}`, 'info');
  }
}

// Inside StaffCard or the Table Row component

const [isRevealed, setIsRevealed] = useState(false);

const handleReveal = async (e: React.MouseEvent) => {
  e.stopPropagation();
  setIsRevealed(true);
  
  // Fire-and-forget tracking
  await apiRequest('POST', '/api/track/reveal', {
    contactName: member.name,
    schoolId: member.schoolId,
    schoolName: member.schoolName
  });
};

// ... In the JSX where {member.email} is displayed ...

<div className="flex items-center gap-2 mt-1">
  <Mail className="h-3 w-3 text-muted-foreground" />
  {isRevealed ? (
    <a href={`mailto:${member.email}`} className="text-primary hover:underline text-xs">
      {member.email}
    </a>
  ) : (
    <button 
      onClick={handleReveal}
      className="text-xs text-muted-foreground bg-muted/50 px-2 py-0.5 rounded hover:bg-muted transition-colors cursor-pointer flex items-center gap-1"
