Step 3: Build the "Master QA Runner"
Create tests/system-audit.ts. This is a standalone script that runs a battery of tests against your running server and codebase.

// tests/system-audit.ts
import { db } from "../server/db";
import { aiExtractFallback } from "../server/lib/ai-extractor";
import { GOLDEN_SET } from "./golden-data";
import { storage } from "../server/storage";

async function runAudit() {
  console.log("🚀 STARTING DEEP SYSTEM AUDIT...");
  const results = { passed: 0, failed: 0 };

  // --- TEST 1: AI & PARSER ACCURACY (Evals) ---
  console.log("\n🧪 1. Running AI/Parser Evals...");
  for (const case of GOLDEN_SET) {
    const start = Date.now();
    // Simulate the extraction logic here
    const result = await aiExtractFallback(case.html, "test-url"); 
    const duration = Date.now() - start;
    
    // Simple equality check (In prod, use fuzzy matching)
    const nameMatch = result?.name === case.expected.name;
    const emailMatch = result?.email === case.expected.email;
    
    if (nameMatch && emailMatch) {
      console.log(`✅ [PASS] ${case.name} (${duration}ms)`);
      results.passed++;
    } else {
      console.error(`❌ [FAIL] ${case.name}`);
      console.error(`   Expected:`, case.expected);
      console.error(`   Got:`, result);
      results.failed++;
    }
  }

  // --- TEST 2: DATABASE INTEGRITY ---
  console.log("\n🗄️ 2. Checking Database Health...");
  // Check for duplicate emails in staff_members (Critical Data Quality issue)
  const duplicates = await db.execute(sql`
    SELECT email, COUNT(*) 
    FROM staff_members 
    WHERE email IS NOT NULL 
    GROUP BY email 
    HAVING COUNT(*) > 1
  `);
  
  if (duplicates.rowCount === 0) {
    console.log("✅ [PASS] No Duplicate Emails found.");
    results.passed++;
  } else {
    console.error(`❌ [FAIL] Found ${duplicates.rowCount} duplicate emails.`);
    results.failed++;
  }

  // --- TEST 3: MODULE LOADING & DEPENDENCY CHECK ---
  console.log("\n📦 3. Verifying Critical Modules...");
  try {
    const playwright = await import("playwright");
    const cheerio = await import("cheerio");
    if (playwright && cheerio) {
      console.log("✅ [PASS] Scraper Modules loaded successfully.");
      results.passed++;
    }
  } catch (e) {
    console.error("❌ [FAIL] Module Load Error:", e);
    results.failed++;
  }

  // --- TEST 4: JOB QUEUE CONCURRENCY SIMULATION ---
  console.log("\n⚡ 4. Stress Testing Job Queue...");
  // Simulate 100 rapid requests to the queue
  const startLoad = Date.now();
  const promises = [];
  for(let i=0; i<50; i++) {
    // We mock the queue push function to measure pure overhead
    promises.push(storage.createExtractionJob({ type: 'test', status: 'pending' }));
  }
  await Promise.all(promises);
  const loadTime = Date.now() - startLoad;
  
  if (loadTime < 2000) {
    console.log(`✅ [PASS] Enqueued 50 jobs in ${loadTime}ms (High Throughput).`);
    results.passed++;
  } else {
    console.warn(`⚠️ [WARN] Job Queue Slow: ${loadTime}ms`);
  }

  // --- FINAL REPORT ---
  console.log("\n==================================");
  console.log(`AUDIT COMPLETE: ${results.passed} Passed, ${results.failed} Failed`);
  if (results.failed > 0) process.exit(1);
  process.exit(0);
}

runAudit();
